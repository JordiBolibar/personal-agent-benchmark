### T8 [ODINN-ecosystem]
Below is a simplified 1D version of the kind of forward pass and manual VJP used in a `SIA2D!` adjoint. Faces k = 1:N−1 lie between cells k and k+1, and end cells are frozen. `vjp_H!` should return ∂(λᵀ dH)/∂H at the state `H` it receives, during an adjoint solve running backward in time.

```julia
function rhs!(dH, H, p, t)
    (; B, Γ, n, Δx, c) = p
    S = B .+ H
    c.dS .= diff(S) ./ Δx
    c.Hf .= (H[1:end-1] .+ H[2:end]) ./ 2
    c.D .= Γ .* c.Hf .^ (n + 2) .* abs.(c.dS) .^ (n - 1)
    q = -c.D .* c.dS
    dH[2:end-1] .= -diff(q) ./ Δx
    dH[1] = dH[end] = 0.0
end

function vjp_H!(out, λ, H, p, t)
    (; Γ, n, Δx, c) = p
    q̄ = diff(λ) ./ Δx
    dqdS = -c.D
    dqdHf = -(n + 2) .* c.D ./ c.Hf .* c.dS
    out .= 0.0
    for k in eachindex(q̄)
        s = q̄[k] * dqdS[k] / Δx
        h = q̄[k] * dqdHf[k] / 2
        out[k]   += h + s
        out[k+1] += h - s
    end
end
```
